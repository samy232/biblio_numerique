from django.http import JsonResponse, HttpResponse
from django.views.decorators.csrf import csrf_exempt
from django.shortcuts import get_object_or_404
from django.views import View
from django.utils.decorators import method_decorator
import json

from .models import Auteur, Livre, Emprunt


# --- Routes Supplémentaires ---

def livres_par_auteur(request, auteur_id):
    """Retourne tous les livres d'un auteur spécifique"""
    # On filtre les livres par l'ID de l'auteur
    livres = Livre.objects.filter(auteur_id=auteur_id)
    data = [
        {
            'id': l.id,
            'titre': l.titre,
            'isbn': l.isbn,
            'disponible': l.disponible
        }
        for l in livres
    ]
    return JsonResponse(data, safe=False)

def emprunts_par_utilisateur(request, nom):
    """Retourne tous les emprunts d'un lecteur par son nom"""
    # On filtre les emprunts par le nom du lecteur
    emprunts = Emprunt.objects.filter(nom_lecteur__icontains=nom)
    data = [
        {
            'id': e.id,
            'livre_titre': e.livre.titre,
            'date_emprunt': str(e.date_emprunt),
            'retourne': e.retourne
        }
        for e in emprunts
    ]
    return JsonResponse(data, safe=False)

# LISTE ET CRÉATION DES EMPRUNTS
# ============================================

@method_decorator(csrf_exempt, name='dispatch')
class ListeEmpruntsView(View):

    def get(self, request):
        """Retourne tous les emprunts"""
        emprunts = Emprunt.objects.all()
        data = [
            {
                'id': e.id,
                'nom_lecteur': e.nom_lecteur,
                'livre_id': e.livre_id,
                'livre_titre': e.livre.titre,
                'date_emprunt': str(e.date_emprunt),
                'date_retour': str(e.date_retour),
                'retourne': e.retourne
            }
            for e in emprunts
        ]
        return JsonResponse(data, safe=False)

    def post(self, request):
        """Crée un nouvel emprunt"""
        try:
            data = json.loads(request.body)
            e = Emprunt.objects.create(
                nom_lecteur=data.get('nom_lecteur'),
                livre_id=data.get('livre_id'),
                date_emprunt=data.get('date_emprunt'),
                date_retour=data.get('date_retour'),
                retourne=False
            )
            return JsonResponse({'id': e.id, 'message': 'Emprunt créé avec succès'}, status=201)
        except Exception as exc:
            return JsonResponse({'erreur': str(exc)}, status=400)

# DETAIL, RETOUR ET SUPPRESSION
# ============================================

@method_decorator(csrf_exempt, name='dispatch')
class DetailEmpruntView(View):

    def get(self, request, id):
        """Détail d’un emprunt"""
        e = get_object_or_404(Emprunt, id=id)
        data = {
            'id': e.id,
            'nom_lecteur': e.nom_lecteur,
            'livre_id': e.livre_id,
            'livre_titre': e.livre.titre,
            'date_emprunt': str(e.date_emprunt),
            'date_retour': str(e.date_retour),
            'retourne': e.retourne
        }
        return JsonResponse(data)

    def put(self, request, id):
        """Marque l’emprunt comme retourné"""
        e = get_object_or_404(Emprunt, id=id)
        e.retourne = True
        e.save()
        return JsonResponse({'message': 'Emprunt marqué comme retourné'})

    def delete(self, request, id):
        """Supprime l’emprunt"""
        e = get_object_or_404(Emprunt, id=id)
        e.delete()
        return JsonResponse({'message': 'Emprunt supprimé'}, status=204)

@csrf_exempt
def creer_auteur(request):
    if request.method != 'POST':
        return JsonResponse({'erreur': 'Méthode non autorisée'}, status=405)

    try:
        data = json.loads(request.body)

        auteur = Auteur.objects.create(
            nom=data.get('nom'),
            prenom=data.get('prenom'),
            date_naissance=data.get('date_naissance'),
            nationalite=data.get('nationalite', '')
        )

        return JsonResponse({
            'id': auteur.id,
            'message': 'Auteur créé avec succès'
        }, status=201)

    except Exception as e:
        return JsonResponse({'erreur': str(e)}, status=400)


@csrf_exempt
def modifier_auteur(request, id):
    if request.method != 'PUT':
        return JsonResponse({'erreur': 'Méthode non autorisée'}, status=405)

    auteur = get_object_or_404(Auteur, id=id)

    try:
        data = json.loads(request.body)

        auteur.nom = data.get('nom', auteur.nom)
        auteur.prenom = data.get('prenom', auteur.prenom)
        auteur.date_naissance = data.get('date_naissance', auteur.date_naissance)
        auteur.nationalite = data.get('nationalite', auteur.nationalite)

        auteur.save()

        return JsonResponse({
            'message': 'Auteur modifié avec succès'
        })

    except Exception as e:
        return JsonResponse({'erreur': str(e)}, status=400)


@csrf_exempt
def supprimer_auteur(request, id):
    if request.method != 'DELETE':
        return JsonResponse({'erreur': 'Méthode non autorisée'}, status=405)

    auteur = get_object_or_404(Auteur, id=id)
    auteur.delete()

    return JsonResponse({
        'message': 'Auteur supprimé avec succès'
    }, status=204)
